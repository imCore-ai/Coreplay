# A/B compare, as in RV: A = the playing clip, B = another playlist item, on one
# frame-locked timeline. Red (A) and blue (B) clips make every mode checkable by colour.
# ---- image sequences
open $MEDIA/cmp_red/red.0001.png|$MEDIA/cmp_blue/blue.0001.png|$MEDIA/png_seq/comp.0001.png|$MEDIA/png_seq/comp.0001.png
until engine == seq 3000
until paused == false 2000
key Space
until paused == true 2000
wait 300
pixel 0.5 0.5 red
expectpixel red == 192,32,32 40
# Ctrl+Shift+A: compare with the next playlist item (Wipe is the first mode)
key Ctrl+Shift+A
until compare.on == true 2000
expect compare.b == blue.0001.png
expect compare.mode == Wipe
until compare.ready == true 3000
wait 300
pixel 0.25 0.5 left
pixel 0.75 0.5 right
expectpixel left == red 12
expectpixel right == 32,32,192 40
expectpixel right != red 60
# \ swaps the sides
key \
until compare.swap == true 1000
wait 300
pixel 0.25 0.5 left2
expectpixel left2 == right 12
key \
until compare.swap == false 1000
# drag the wipe line to the right: 0.75 is A now
ldrag 0.5 0.5 0.9 0.5
until compare.wipe > 0.8 1000
expect paused == true
wait 300
pixel 0.75 0.5 right3
expectpixel right3 == red 12
# F6 flip: A whole, \ shows B whole
key F6
until compare.mode == A/B Flip 1000
wait 300
pixel 0.5 0.5 flipA
expectpixel flipA == red 12
key \
wait 300
pixel 0.5 0.5 flipB
expectpixel flipB == right 12
key \
# F8 side by side: A on the left half of the view, B on the right
key F8
until compare.mode == Side by Side 1000
wait 300
pixel 0.25 0.5 sbsL
pixel 0.75 0.5 sbsR
expectpixel sbsL == red 12
expectpixel sbsR == right 12
# F9 difference: |red - blue|
key F9
until compare.mode == Difference 1000
wait 300
pixel 0.5 0.5 diff
expectpixel diff == 160,0,160 40
# B frame offset: Alt+Shift+Right / Left
key Home
until frame == 1 2000
expect compare.bFrame == 1
key Alt+Shift+Right
key Alt+Shift+Right
until compare.offset == 2 1000
expect compare.bFrame == 3
key Alt+Shift+Left
until compare.offset == 1 1000
key Ctrl+Shift+A
until compare.on == false 2000
expect compare.mode == Off
# frame-locked while playing: a sequence against itself is black in Difference
key Ctrl+PgDown
key Ctrl+PgDown
until title == comp.####.png [1-48] - Coreplay 3000
compare start 3
until compare.on == true 2000
key F9
until paused == false 2000
wait 500
expect paused == false
pixel 0.3 0.4 lockA
wait 300
pixel 0.6 0.6 lockB
expectpixel lockA == 0,0,0 24
expectpixel lockB == 0,0,0 24
expect frame > 12
