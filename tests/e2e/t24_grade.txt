# Colour grading (Alt+G): the grade shows live on both engines (GPU), the clip keeps
# playing, the title marks the clip modified, and Save As / Save / Export write the
# same colours (baked .cube) as the player shows.
open $MEDIA/patches_709.mp4
until engine == mpv 3000
until media.w > 0 3000
wait 400
pixel 0.125 0.25 m_raw_0
pixel 0.375 0.25 m_raw_1
pixel 0.625 0.25 m_raw_2
pixel 0.875 0.25 m_raw_3

pixel 0.375 0.75 m_raw_5
pixel 0.625 0.75 m_raw_6
pixel 0.875 0.75 m_raw_7
expect grade.modified == false
expect grade.window == false
key Alt+G
until grade.window == true 2000
grade grab $OUT/t24_grade_window.png
# the window moves by its title bar, and keeps its place when it opens again
grade drag -300 80
grade drag 120 -40
key Alt+G
until grade.window == false 2000
key Alt+G
until grade.window == true 2000
grade drag 0 0
# every change shows, not only the first two (mpv keeps a shader file it read once, by name)

wait 300
grade set brightness 102
wait 300
grade set brightness 150
wait 300
pixel 0.375 0.25 m_b150
expectpixel m_b150 != m_raw_1 20

wait 300
