# Export / transcode: movie and sequence sources to video, image sequence and audio formats.
# Every output is checked with ffprobe: codec, exact frame rate (r AND avg), frame count, duration.
# --- movie source: 10 s H.264 at 24000/1001 with AAC audio (240 frames)
open $MEDIA/h264_23976.mp4
until fps.rational == 24000/1001 5000
export h264 $OUT/exp/m_h264.mp4
until export.state != running 60000
expect export.state == done
probe $OUT/exp/m_h264.mp4 codec == h264
probe $OUT/exp/m_h264.mp4 r_frame_rate == 24000/1001
probe $OUT/exp/m_h264.mp4 avg_frame_rate == 24000/1001
probe $OUT/exp/m_h264.mp4 frames == 240
probe $OUT/exp/m_h264.mp4 audio_codec == aac
export h264_nvenc $OUT/exp/m_nvenc.mp4
until export.state != running 60000
expect export.state == done
probe $OUT/exp/m_nvenc.mp4 codec == h264
probe $OUT/exp/m_nvenc.mp4 frames == 240
probe $OUT/exp/m_nvenc.mp4 avg_frame_rate == 24000/1001
export prores422hq $OUT/exp/m_prores.mov
until export.state != running 60000
expect export.state == done
probe $OUT/exp/m_prores.mov codec == prores
probe $OUT/exp/m_prores.mov pix_fmt == yuv422p10le
probe $OUT/exp/m_prores.mov frames == 240
probe $OUT/exp/m_prores.mov duration ~ 10.01
probe $OUT/exp/m_prores.mov audio_codec == pcm_s24le
export dnxhr_hqx $OUT/exp/m_dnxhr.mov
until export.state != running 60000
expect export.state == done
probe $OUT/exp/m_dnxhr.mov codec == dnxhd
probe $OUT/exp/m_dnxhr.mov frames == 240
probe $OUT/exp/m_dnxhr.mov avg_frame_rate == 24000/1001
export ffv1 $OUT/exp/m_ffv1.mkv
until export.state != running 60000
expect export.state == done
probe $OUT/exp/m_ffv1.mkv codec == ffv1
probe $OUT/exp/m_ffv1.mkv frames == 240
probe $OUT/exp/m_ffv1.mkv audio_codec == flac
export h265 $OUT/exp/m_h265.mp4
until export.state != running 120000
expect export.state == done
probe $OUT/exp/m_h265.mp4 codec == hevc
probe $OUT/exp/m_h265.mp4 frames == 240
export wav $OUT/exp/m_audio.wav
until export.state != running 60000
expect export.state == done
probe $OUT/exp/m_audio.wav audio_codec == pcm_s24le
probe $OUT/exp/m_audio.wav duration ~ 10.0
# Range 2 s .. 4 s of a 24000/1001 movie = 48 frames, to a PNG sequence
export png8 $OUT/exp/m_png/m.%04d.png range=2.002:4.004
until export.state != running 60000
expect export.state == done
probe $OUT/exp/m_png/m.1001.png exists == yes
probe $OUT/exp/m_png/m.1048.png exists == yes
probe $OUT/exp/m_png/m.1049.png exists == no
# The name the Export window suggests for a sequence (it once came out as "%pngd" and
# every frame went to one file: "Writing packet failed")
export png16 suggest=$OUT/exp range=2.002:4.004
until export.state != running 60000
expect export.state == done
expect export.suggested == $OUT/exp/h264_23976_png16/h264_23976.%04d.png
probe $OUT/exp/h264_23976_png16/h264_23976.1001.png exists == yes
probe $OUT/exp/h264_23976_png16/h264_23976.1048.png exists == yes
# Names the user types: "####" (Nuke / RV style), no frame number at all, and a "%" in
# the name. Each must still give one file per frame, never one file for every frame.
export png8 $OUT/exp/hash/h_####.png range=2.002:2.169
until export.state != running 60000
expect export.state == done
probe $OUT/exp/hash/h_1001.png exists == yes
probe $OUT/exp/hash/h_1004.png exists == yes
export png8 $OUT/exp/plain/p.png range=2.002:2.169
until export.state != running 60000
expect export.state == done
probe $OUT/exp/plain/p.1001.png exists == yes
probe $OUT/exp/plain/p.1004.png exists == yes
export png8 $OUT/exp/pct/50%_grey.%04d.png range=2.002:2.169
until export.state != running 60000
expect export.state == done
probe $OUT/exp/pct/50%_grey.1001.png exists == yes
probe $OUT/exp/pct/50%_grey.1004.png exists == yes
export prores422hq $OUT/exp/m_range.mov range=2.002:4.004
until export.state != running 60000
probe $OUT/exp/m_range.mov frames == 48
probe $OUT/exp/m_range.mov duration ~ 2.002
